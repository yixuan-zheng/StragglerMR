cmake_minimum_required(VERSION 3.24)
project(StragglerMR VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(Protobuf CONFIG QUIET)
if(NOT Protobuf_FOUND)
  # Some Linux distributions provide only CMake's FindProtobuf module.
  find_package(Protobuf REQUIRED)
endif()
find_package(gRPC CONFIG REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
option(STRAGGLERMR_SANITIZERS "Enable selected sanitizers" OFF)
set(STRAGGLERMR_SANITIZER_SET "address,undefined" CACHE STRING "Clang sanitizer set")
set(PROTO ${CMAKE_CURRENT_SOURCE_DIR}/proto/runtime.proto)
set(GEN ${CMAKE_CURRENT_BINARY_DIR}/generated)
file(MAKE_DIRECTORY ${GEN})
add_custom_command(OUTPUT ${GEN}/runtime.pb.cc ${GEN}/runtime.pb.h ${GEN}/runtime.grpc.pb.cc ${GEN}/runtime.grpc.pb.h
  COMMAND protobuf::protoc ARGS --proto_path=${CMAKE_CURRENT_SOURCE_DIR}/proto --cpp_out=${GEN}
    --grpc_out=${GEN} --plugin=protoc-gen-grpc=$<TARGET_FILE:gRPC::grpc_cpp_plugin> ${PROTO}
  DEPENDS ${PROTO} protobuf::protoc gRPC::grpc_cpp_plugin VERBATIM)
add_library(mr_protocol ${GEN}/runtime.pb.cc ${GEN}/runtime.grpc.pb.cc)
target_include_directories(mr_protocol PUBLIC ${GEN})
target_link_libraries(mr_protocol PUBLIC gRPC::grpc++ protobuf::libprotobuf)
add_library(mr_common src/common.cpp)
target_include_directories(mr_common PUBLIC src)
target_link_libraries(mr_common PUBLIC Threads::Threads OpenSSL::Crypto)
foreach(target mr_worker mr_coordinator)
  string(REPLACE "mr_" "" source ${target})
  add_executable(${target} src/${source}.cpp)
  target_link_libraries(${target} PRIVATE mr_common mr_protocol)
endforeach()
add_executable(mr_protocol_probe tests/protocol_probe.cpp)
target_link_libraries(mr_protocol_probe PRIVATE mr_common mr_protocol)
add_executable(mr_validation_test tests/validation.cpp)
target_link_libraries(mr_validation_test PRIVATE mr_common)
foreach(target mr_common mr_worker mr_coordinator mr_protocol_probe mr_validation_test)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(STRAGGLERMR_SANITIZERS)
    target_compile_options(${target} PUBLIC -fsanitize=${STRAGGLERMR_SANITIZER_SET} -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=${STRAGGLERMR_SANITIZER_SET})
  endif()
endforeach()
enable_testing()
add_test(NAME artifact_validation COMMAND mr_validation_test)
foreach(case healthy1 healthy2 healthy4 capacity retry_error malformed missing crash late lost_reply exhausted unavailable input_error oversized_input saturated_timeout publication_error protocol)
  add_test(NAME ${case} COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py
    --bin-dir $<TARGET_FILE_DIR:mr_worker> --case ${case})
  set_tests_properties(${case} PROPERTIES TIMEOUT 30)
endforeach()
foreach(case backup_wins original_wins backup_fails original_crashes idle_unavailable pending_priority concurrent_limit attempt_budget exhausted_budget malformed_backup corrupt_backup lost_reply insufficient_peers incomparable_peers no_spare cleanup)
  add_test(NAME spec_${case} COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/speculation.py
    --bin-dir $<TARGET_FILE_DIR:mr_worker> --case ${case})
  set_tests_properties(spec_${case} PROPERTIES TIMEOUT 30)
endforeach()
